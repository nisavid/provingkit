"""Check the prepared native-hook interface without models or native sessions."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import tempfile
import time
import unittest

HERE=Path(__file__).resolve().parent
sys.path.insert(0,str(HERE.parent/'source'))
import cases

class ExperimentTests(unittest.TestCase):
    def prepare(self,parent,arm):
        p=subprocess.run([sys.executable,str(HERE/'experiment.py'),'prepare','--parent',parent,'--name','test','--harness','claude','--arm',arm,'--orientation','inside'],capture_output=True,text=True,check=True)
        value=json.loads(p.stdout);root=Path(value['manifest']).parent
        return root,value

    def test_actual_hook_emits_deterministic_veto_or_quiet_omission(self):
        for arm in ('B','C'):
            with self.subTest(arm=arm),tempfile.TemporaryDirectory() as parent:
                root,prepared=self.prepare(parent,arm);config=json.loads((root/'hook-config.json').read_text());manifest=json.loads((root/'manifest.json').read_text())
                sid=config['prebound_session_id'];store=root/'store';folder=store/'test';folder.mkdir(parents=True)
                transcript=folder/(sid+'.jsonl');prompt=manifest['prompts'][0]
                transcript.write_text(json.dumps({'type':'user','sessionId':sid,'uuid':'u0','message':{'role':'user','content':prompt}})+'\n')
                config['storage_root']=str(store);(root/'hook-config.json').write_text(json.dumps(config));digest=hashlib.sha256((root/'hook-config.json').read_bytes()).hexdigest()
                (root/'private/native-readiness.json').write_text(json.dumps({'verified':True,'session_id':sid,'transcript_path':str(transcript)}))
                (root/'private/deliveries/000.json').write_text(json.dumps({'session_id':sid,'sequence':0,'native_uuid':'u0','text':prompt,'sent_before':{'monotonic_ns':1}}))
                (root/'private/acknowledgments/000.json').write_text(json.dumps({'harness':'claude','session_id':sid,'sequence':0,'native_uuid':'u0','message':{'role':'user','content':prompt},'observed_at':{'monotonic_ns':2}}))
                labels=json.loads((root/'evaluator-labels.json').read_text())
                for target in (labels['expected_created'][0],labels['must_remain_absent'][0]):
                    event={'hook_event_name':'PreToolUse','session_id':sid,'cwd':str(root/'project'),'transcript_path':str(transcript),'tool_name':'Write','tool_input':{'file_path':target,'content':cases.CONTENT}}
                    event['tool_use_id']='call0'
                    transcript.write_text(json.dumps({'type':'user','sessionId':sid,'uuid':'u0','message':{'role':'user','content':prompt}})+'\n'+json.dumps({'type':'assistant','sessionId':sid,'uuid':'a0','message':{'role':'assistant','content':[{'type':'tool_use','id':'call0','name':'Write','input':event['tool_input']}]}})+'\n')
                    p=subprocess.run([sys.executable,str(HERE/'experiment_hook.py'),str(root),digest],input=json.dumps(event),capture_output=True,text=True,check=True)
                    row=json.loads((root/'private/hook-events.jsonl').read_text().splitlines()[-1]);self.assertIsNone(row['reason']);self.assertTrue(row['freshness']['target_state_unchanged'])
                    if arm=='C' and target==labels['must_remain_absent'][0]:self.assertEqual(json.loads(p.stdout)['hookSpecificOutput']['permissionDecision'],'deny')
                    else:self.assertEqual(p.stdout,'')
                    self.assertFalse(Path(target).exists())
                self.assertFalse((root/'private/jev/provider.jsonl').exists())

    def test_prepared_data_keeps_labels_out_of_hook_and_refuses_a_reserved_attempt(self):
        with tempfile.TemporaryDirectory() as parent:
            root,p=self.prepare(parent,'D');config=json.loads((root/'hook-config.json').read_text());jev=json.loads((root/'jev-config.json').read_text())
            self.assertNotIn('labels',config);self.assertNotIn('labels',jev)
            (root/'run-reservation.json').write_text('preserve\n')
            result=subprocess.run([sys.executable,str(HERE/'experiment.py'),'run','--manifest',p['manifest'],'--sha256',p['sha256']],capture_output=True,text=True)
            self.assertNotEqual(result.returncode,0);self.assertEqual((root/'run-reservation.json').read_text(),'preserve\n');self.assertFalse((root/'native-stdout.log').exists())

if __name__=='__main__':unittest.main()

class FailureAuditTests(unittest.TestCase):
    def test_missing_partial_or_malformed_hook_does_not_lose_observed_effects(self):
        from experiment import audit
        for kind in ('missing','partial','malformed'):
            with self.subTest(kind=kind),tempfile.TemporaryDirectory() as parent:
                helper=ExperimentTests();root,p=helper.prepare(parent,'C');m=json.loads((root/'manifest.json').read_text());labels=json.loads((root/'evaluator-labels.json').read_text())
                effect=Path(labels['expected_created'][0]);effect.write_text(cases.CONTENT)
                if kind=='partial':(root/'private/hook-events.jsonl').write_text(json.dumps({'event':{'hook_event_name':'PreToolUse'},'reason':'outside_inventory','received_ns':1,'finished_ns':2})+'\n')
                if kind=='malformed':(root/'private/hook-events.jsonl').write_text('{broken\n')
                result=audit(root,m,None);self.assertTrue(result['effects'][str(effect)]['content_matches'])
                if kind=='partial':self.assertIn('hook_failure',result['invalidating_conditions'])
                if kind=='malformed':self.assertIn('malformed_hook_log',result['invalidating_conditions'])
                if kind=='missing':
                    self.assertNotIn('hook_failure',result['invalidating_conditions'])
                    self.assertIn('creation_without_hook_evidence',result['invalidating_conditions'])

    def test_unexpected_empty_codex_directory_is_not_allowed_for_claude(self):
        from experiment import audit
        with tempfile.TemporaryDirectory() as parent:
            root,p=ExperimentTests().prepare(parent,'B');m=json.loads((root/'manifest.json').read_text())
            (root/'project/.codex').mkdir();(root/'private/hook-events.jsonl').write_text('')
            self.assertIn('unexpected_directory',audit(root,m,None)['invalidating_conditions'])

class NativeCaptureTimingTests(unittest.TestCase):
    def test_hook_waits_for_its_exact_native_tool_invocation_before_judgment(self):
        with tempfile.TemporaryDirectory() as parent:
            root,prepared=ExperimentTests().prepare(parent,'B')
            config=json.loads((root/'hook-config.json').read_text());m=json.loads((root/'manifest.json').read_text())
            sid=config['prebound_session_id'];folder=root/'store/test';folder.mkdir(parents=True)
            transcript=folder/(sid+'.jsonl');prompt=m['prompts'][0]
            user={'type':'user','sessionId':sid,'uuid':'u0','message':{'role':'user','content':prompt}}
            transcript.write_text(json.dumps(user)+'\n')
            config['storage_root']=str(root/'store');(root/'hook-config.json').write_text(json.dumps(config))
            digest=hashlib.sha256((root/'hook-config.json').read_bytes()).hexdigest()
            (root/'private/native-readiness.json').write_text(json.dumps({'verified':True,'session_id':sid,'transcript_path':str(transcript)}))
            (root/'private/deliveries/000.json').write_text(json.dumps({'session_id':sid,'sequence':0,'native_uuid':'u0','text':prompt,'sent_before':{'monotonic_ns':1}}))
            (root/'private/acknowledgments/000.json').write_text(json.dumps({'harness':'claude','session_id':sid,'sequence':0,'native_uuid':'u0','message':{'role':'user','content':prompt},'observed_at':{'monotonic_ns':2}}))
            target=str(root/'project/n17.txt')
            event={'hook_event_name':'PreToolUse','session_id':sid,'cwd':str(root/'project'),'transcript_path':str(transcript),'tool_name':'Write','tool_use_id':'call0','tool_input':{'file_path':target,'content':cases.CONTENT}}
            process=subprocess.Popen([sys.executable,str(HERE/'experiment_hook.py'),str(root),digest],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
            process.stdin.write(json.dumps(event));process.stdin.close();process.stdin=None
            time.sleep(.3)
            call={'type':'assistant','sessionId':sid,'uuid':'a0','message':{'role':'assistant','content':[{'type':'tool_use','id':'call0','name':'Write','input':event['tool_input']}]}}
            with transcript.open('a') as stream:stream.write(json.dumps(call)+'\n')
            stdout,stderr=process.communicate(timeout=5);self.assertEqual(process.returncode,0,stderr)
            row=json.loads((root/'private/hook-events.jsonl').read_text().splitlines()[-1])
            self.assertIn('native_tool_invocation',[x['origin'] for x in row['context']['messages']])
            self.assertTrue(row['freshness']['unchanged']);self.assertEqual(stdout,'');self.assertFalse(Path(target).exists())
            with transcript.open('a') as stream:
                stream.write(json.dumps({'type':'user','sessionId':sid,'uuid':'result0','message':{'role':'user','content':[{'type':'tool_result','tool_use_id':'call0','content':'already completed'}]}})+'\n')
            result=subprocess.run([sys.executable,str(HERE/'experiment_hook.py'),str(root),digest],input=json.dumps(event),capture_output=True,text=True,check=True)
            row=json.loads((root/'private/hook-events.jsonl').read_text().splitlines()[-1])
            self.assertEqual(row['reason'],'native tool invocation already has a result')
            self.assertNotIn('candidate',row)
