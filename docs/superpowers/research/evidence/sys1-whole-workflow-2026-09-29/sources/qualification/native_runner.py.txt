"""One-turn native qualification controller; the caller owns reservation and outcome review."""

import hashlib
import json
import os
import time
import uuid
from pathlib import Path

from native_support import (MODELS, VERSIONS, codex_common, discover_codex_hooks,
                            require, verify_claude_init, verify_codex_sandbox)
from observer import stamp
from transport import collect, encoded, put


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def atomic_private(root, name, value):
    """Publish one complete, exclusive private record without replacing an earlier fact."""
    target = root / "private" / name
    temporary = root / "private" / ("." + name + "." + uuid.uuid4().hex)
    try:
        put(temporary, encoded(value))
        os.link(temporary, target)
    finally:
        temporary.unlink(missing_ok=True)


def checked_manifest(manifest, root):
    require(manifest.get("schema") == "sys1-native-provenance-qualification/v1",
            "manifest schema differs")
    harness = manifest.get("harness")
    require(harness in MODELS and manifest.get("version") == VERSIONS[harness]
            and manifest.get("model") == MODELS[harness] and manifest.get("effort") == "high",
            "native selection differs")
    project, sibling = Path(manifest["project"]), Path(manifest["sibling"])
    require(root.is_absolute() and root.resolve() == root and project == root / "project"
            and sibling == root / "sibling" and project.resolve() == project
            and sibling.resolve() == sibling and project.is_dir() and sibling.is_dir(),
            "fixture roots differ")
    require(manifest.get("deadline_seconds") == 180, "deadline differs")
    require(manifest.get("inventory") == {str(sibling / "n17.txt"): "SYS1_SYNTHETIC_FILE\n"},
            "fixture inventory differs")
    settings = Path(manifest["settings_path"])
    expected = root / "settings.json" if harness == "claude" else project / ".codex" / "hooks.json"
    require(settings == expected and digest(settings) == manifest["settings_sha256"],
            "native hook settings drift")
    require(digest(root / "hook-config.json") == manifest["hook_config_sha256"],
            "hook config drift")
    require(all(digest(Path(path)) == value for path, value in manifest["source_sha256"].items()),
            "source drift")
    require(not any((root / "private" / name).exists() for name in
                    ("native-readiness.json", "delivery.json", "acknowledgment.json")),
            "native attempt already has controller records")
    return harness, project, sibling


def bounded_collect(command, project, root, label, deadline, **callbacks):
    return collect(command, project, root, label, deadline, **callbacks)


def check_version(harness, manifest, project, root, deadline):
    output, code = bounded_collect([harness, "--version"], project, root, "version",
                                   min(deadline, time.monotonic() + 10))
    require(code == 0 and output.decode().strip() == manifest["version"],
            "native CLI version differs")


def run_claude(manifest, root, project, sibling, deadline):
    command = ["claude", "-p", "--restricted", "--tools", "Write",
               "--setting-sources", "", "--settings", str(root / "settings.json"),
               "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
               "--disable-slash-commands", "--permission-mode", "auto",
               "--permission-prompts", "none", "--add-dir", str(sibling),
               "--model", MODELS["claude"], "--effort", "high",
               "--session-id", manifest["claude_session_id"],
               "--input-format", "stream-json", "--output-format", "stream-json",
               "--replay-user-messages", "--include-hook-events", "--verbose",
               "--system-prompt", "Perform only the user's bounded synthetic file task. "
               "Use Write for one file. Do not delegate."]
    session = manifest["claude_session_id"]
    require(isinstance(session, str) and uuid.UUID(session).version == 4,
            "Claude prebound session differs")
    envelope = {"type": "user", "uuid": str(uuid.uuid4()), "session_id": "",
                "parent_tool_use_id": None,
                "message": {"role": "user", "content": manifest["prompt"]}}
    state = {"control": False, "init": False, "sent": False, "ack": False,
             "result": None, "events": [], "tool_ids": set(), "tool_results": set()}

    def maybe_send(send):
        if not state["control"] or state["sent"]:
            return
        atomic_private(root, "delivery.json",
                       {"schema": "sys1-native-delivery/v1", "harness": "claude",
                        "session_id": session, "sequence": 0, "text": manifest["prompt"],
                        "native_uuid": envelope["uuid"], "envelope": envelope,
                        "sent_before": stamp()})
        state["sent"] = True
        send(envelope)

    def begin(send):
        send({"type": "control_request", "request_id": "sys1-native-initialize",
              "request": {"subtype": "initialize"}})

    def receive(event, send):
        state["events"].append(event)
        kind = event.get("type")
        if kind == "control_request":
            raise RuntimeError("native permission/control request requires adjudication")
        if kind == "control_response":
            response = event.get("response", {})
            require(response.get("request_id") == "sys1-native-initialize"
                    and response.get("subtype") == "success" and not state["control"],
                    "Claude control initialization differs")
            models = response.get("response", {}).get("models", [])
            matches = [item for item in models if item.get("value") == "opus"]
            require(len(matches) == 1 and matches[0].get("resolvedModel") == MODELS["claude"]
                    and "high" in matches[0].get("supportedEffortLevels", []),
                    "Claude Opus/high catalog differs")
            state["control"] = True
            maybe_send(send)
        elif kind == "system" and event.get("subtype") == "init":
            require(not state["init"], "duplicate Claude init")
            verify_claude_init(manifest, event)
            state["init"] = True
            atomic_private(root, "native-readiness.json",
                           {"schema": "sys1-native-readiness/v1", "harness": "claude",
                            "session_id": session, "transcript_path": None, "verified": True,
                            "verified_at": stamp()})
            maybe_send(send)
        elif kind == "assistant":
            require(event.get("session_id") == session, "Claude assistant session differs")
            for part in event.get("message", {}).get("content", []):
                if part.get("type") == "tool_use":
                    tool_id = part.get("id")
                    require(isinstance(tool_id, str) and tool_id and tool_id not in state["tool_ids"],
                            "Claude tool identity differs")
                    state["tool_ids"].add(tool_id)
        elif kind == "user" and event.get("isReplay") is True:
            require(state["sent"] and not state["ack"]
                    and event.get("uuid") == envelope["uuid"]
                    and event.get("message") == envelope["message"]
                    and event.get("session_id") == session,
                    "Claude native user acknowledgment differs")
            atomic_private(root, "acknowledgment.json",
                           {"schema": "sys1-native-acknowledgment/v1", "harness": "claude",
                            "session_id": session, "sequence": 0, "native_uuid": event["uuid"],
                            "message": event["message"], "observed_at": stamp()})
            state["ack"] = True
        elif kind == "user":
            message = event.get("message", {})
            content = message.get("content")
            require(state["ack"] and event.get("session_id") == session
                    and message.get("role") == "user" and isinstance(content, list) and content
                    and all(isinstance(part, dict) and part.get("type") == "tool_result" for part in content),
                    "Claude unbound user message")
            for part in content:
                tool_id = part.get("tool_use_id")
                require(isinstance(tool_id, str) and tool_id in state["tool_ids"]
                        and tool_id not in state["tool_results"], "Claude unbound tool result")
                state["tool_results"].add(tool_id)
        elif kind == "result":
            require(state["sent"] and state["init"] and state["ack"] and state["result"] is None
                    and state["tool_ids"] == state["tool_results"]
                    and event.get("session_id") == session and event.get("is_error") is False,
                    "Claude native result differs")
            state["result"] = event
            return True
        return False

    _, exit_code = bounded_collect(command, project, root, "native", deadline,
                                   on_line=receive, begin=begin)
    raw = root / "native-stdout.log"
    require(raw.read_bytes().endswith(b"\n"), "incomplete Claude native stream")
    return {"harness": "claude", "session_id": session,
            "acknowledged_uuid": envelope["uuid"], "native_result": state["result"],
            "raw_event_count": len(state["events"]), "raw_stdout_path": str(raw),
            "raw_stdout_sha256": digest(raw), "native_process_exit_code": exit_code,
            "native_process_shutdown": "transport terminated process group after completion callback"}


def codex_catalog(project, root, deadline):
    output, code = bounded_collect(["codex", "debug", "models"], project, root, "catalog",
                                   min(deadline, time.monotonic() + 30))
    require(code == 0, "Codex model catalog failed")
    models = [item for item in json.loads(output).get("models", [])
              if item.get("slug") == MODELS["codex"]]
    require(len(models) == 1 and any(item.get("effort") == "high"
            for item in models[0].get("supported_reasoning_levels", [])),
            "Codex Sol/high catalog differs")


def run_codex(manifest, root, project, sibling, deadline):
    codex_catalog(project, root, deadline)
    def within_deadline(command, cwd, attempt, label, requested, **callbacks):
        return collect(command, cwd, attempt, label, min(deadline, requested), **callbacks)
    override = discover_codex_hooks(manifest, root, within_deadline)
    command = codex_common(manifest, project, override)
    state = {"phase": "initialize", "session": None, "path": None,
             "turn_id": None, "history": None, "notifications": [], "settings": []}

    def begin(send):
        send({"id": 1, "method": "initialize",
              "params": {"clientInfo": {"name": "sys1-native-probe", "version": "1"}}})

    def receive(message, send):
        method, ident = message.get("method"), message.get("id")
        if method == "thread/settings/updated":
            params = message.get("params", {})
            settings = params.get("threadSettings", {})
            require(params.get("threadId") == state["session"]
                    and settings.get("model") == MODELS["codex"]
                    and settings.get("effort") == "high"
                    and settings.get("approvalPolicy") == "on-request"
                    and settings.get("approvalsReviewer") == "user",
                    "Codex effective settings differ")
            verify_codex_sandbox(settings, project, "settings", sibling)
            state["settings"].append(message)
            return False
        if method == "thread/tokenUsage/updated":
            state["notifications"].append(message)
            return False
        if method is not None and ident is not None:
            raise RuntimeError("native permission/server request requires adjudication")
        if ident == 1:
            require(state["phase"] == "initialize" and "result" in message,
                    "Codex initialize differs")
            state["phase"] = "thread-start"
            send({"method": "initialized"})
            send({"id": 2, "method": "thread/start",
                  "params": {"model": MODELS["codex"], "cwd": str(project),
                             "approvalPolicy": "on-request", "approvalsReviewer": "user",
                             "sandbox": "workspace-write", "ephemeral": False,
                             "config": {"sandbox_workspace_write": {
                                 "writable_roots": [str(sibling)], "network_access": False}}}})
        elif ident == 2:
            require(state["phase"] == "thread-start" and "result" in message,
                    "Codex thread/start failed")
            response = message["result"]
            thread = response["thread"]
            require(thread.get("cliVersion") == "0.159.0" and thread.get("ephemeral") is False
                    and thread.get("turns") == [] and response.get("model") == MODELS["codex"]
                    and response.get("reasoningEffort") == "high"
                    and response.get("cwd") == str(project)
                    and response.get("approvalPolicy") == "on-request"
                    and response.get("approvalsReviewer") == "user",
                    "Codex fresh thread/settings differ")
            verify_codex_sandbox(response, project, "thread-start", sibling)
            session = thread.get("id")
            path = Path(thread.get("path") or "")
            storage = Path(manifest["storage_root"])
            require(isinstance(session, str) and session
                    and path.is_absolute() and path.parent.resolve() == path.parent
                    and path.is_relative_to(storage)
                    and len(path.parent.relative_to(storage).parts) == 3
                    and path.name.endswith("-" + session + ".jsonl"),
                    "Codex native session/transcript identity differs")
            state["session"], state["path"] = session, str(path)
            state["phase"] = "connectors"
            send({"id": 3, "method": "mcpServerStatus/list",
                  "params": {"threadId": session, "limit": 100}})
        elif ident == 3:
            require(state["phase"] == "connectors" and "result" in message,
                    "Codex connector inventory unavailable")
            result = message["result"]
            require(result.get("nextCursor") is None and isinstance(result.get("data"), list)
                    and all(item.get("runtimeStatus") == "disabled"
                            and item.get("tools") == {} and item.get("resources") == []
                            and item.get("resourceTemplates") == []
                            for item in result["data"]), "Codex inherited connector active")
            atomic_private(root, "native-readiness.json",
                           {"schema": "sys1-native-readiness/v1", "harness": "codex",
                            "session_id": state["session"], "transcript_path": state["path"],
                            "verified": True, "verified_at": stamp()})
            atomic_private(root, "delivery.json",
                           {"schema": "sys1-native-delivery/v1", "harness": "codex",
                            "session_id": state["session"], "sequence": 0,
                            "text": manifest["prompt"], "sent_before": stamp()})
            state["phase"] = "turn-start"
            send({"id": 4, "method": "turn/start",
                  "params": {"threadId": state["session"],
                             "input": [{"type": "text", "text": manifest["prompt"],
                                        "text_elements": []}],
                             "model": MODELS["codex"], "effort": "high",
                             "approvalPolicy": "on-request", "approvalsReviewer": "user",
                             "cwd": str(project), "sandboxPolicy": {
                                 "type": "workspaceWrite", "writableRoots": [str(sibling)],
                                 "networkAccess": False}}})
        elif ident == 4:
            require(state["phase"] == "turn-start" and "result" in message,
                    "Codex turn/start failed")
            turn_id = message["result"]["turn"]["id"]
            require(isinstance(turn_id, str) and turn_id, "Codex turn ID unavailable")
            state["turn_id"] = turn_id
            atomic_private(root, "acknowledgment.json",
                           {"schema": "sys1-native-acknowledgment/v1", "harness": "codex",
                            "session_id": state["session"], "sequence": 0,
                            "turn_id": turn_id, "observed_at": stamp()})
            state["phase"] = "running"
        elif method == "turn/completed":
            params = message.get("params", {})
            turn = params.get("turn", {})
            require(state["phase"] == "running"
                    and params.get("threadId") == state["session"]
                    and turn.get("id") == state["turn_id"]
                    and turn.get("status") == "completed" and turn.get("error") is None,
                    "Codex native turn completion differs")
            state["phase"] = "history"
            send({"id": 5, "method": "thread/turns/list",
                  "params": {"threadId": state["session"], "limit": 10,
                             "sortDirection": "asc", "itemsView": "full"}})
        elif ident == 5:
            require(state["phase"] == "history" and "result" in message,
                    "Codex full history unavailable")
            result = message["result"]
            turns = result.get("data")
            require(result.get("nextCursor") is None and isinstance(turns, list)
                    and len(turns) == 1 and turns[0].get("id") == state["turn_id"]
                    and turns[0].get("status") == "completed"
                    and turns[0].get("error") is None
                    and turns[0].get("itemsView") == "full",
                    "Codex completed history differs")
            users = [item for item in turns[0].get("items", [])
                     if item.get("type") == "userMessage"]
            require(len(users) == 1 and users[0].get("content") == [
                {"type": "text", "text": manifest["prompt"], "text_elements": []}],
                "Codex native user content differs")
            state["history"] = turns[0]
            return True
        elif ident is not None:
            raise RuntimeError("unexpected Codex response ID")
        return False

    _, exit_code = bounded_collect(command, project, root, "native", deadline,
                                   on_line=receive, begin=begin)
    raw = root / "native-stdout.log"
    require(raw.read_bytes().endswith(b"\n"), "incomplete Codex native stream")
    return {"harness": "codex", "session_id": state["session"],
            "transcript_path": state["path"], "turn_id": state["turn_id"],
            "native_turn": state["history"], "settings_updates": state["settings"],
            "raw_stdout_path": str(raw), "raw_stdout_sha256": digest(raw),
            "native_process_exit_code": exit_code,
            "native_process_shutdown": "transport terminated process group after completion callback"}


def run(manifest, root):
    """Run one externally reserved qualification; return observations, not authority."""
    root = Path(root)
    harness, project, sibling = checked_manifest(manifest, root)
    deadline = time.monotonic() + 180
    check_version(harness, manifest, project, root, deadline)
    if harness == "claude":
        return run_claude(manifest, root, project, sibling, deadline)
    return run_codex(manifest, root, project, sibling, deadline)
