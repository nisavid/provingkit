import json
from pathlib import Path
import sys
import tempfile
import time
import unittest

from transport import collect


class TransportTests(unittest.TestCase):
    def test_terminal_callback_has_a_byte_cutoff_and_undelivered_output_is_retained(self):
        script = '''import json,os,signal,sys
def stop(*args):
 os.write(1,b'{"late":2}\\n'); sys.exit(0)
signal.signal(signal.SIGTERM,stop)
os.write(1,b'{"terminal":true}\\n{"late":1}\\n')
while True: signal.pause()
'''
        with tempfile.TemporaryDirectory() as name:
            root = Path(name)
            received = []
            def receive(row, send):
                received.append(row)
                return row.get('terminal') is True
            collect([sys.executable, '-c', script], root, root, 'fake', time.monotonic() + 3,
                    on_line=receive, begin=lambda send: None)
            self.assertEqual(received, [{'terminal': True}])
            process = json.loads((root / 'fake-process.json').read_text())
            cutoff = process['terminal_callback_stdout_offset']
            raw = (root / 'fake-stdout.log').read_bytes()
            self.assertEqual(raw[:cutoff], b'{"terminal":true}\n')
            self.assertEqual((root / 'fake-undelivered-stdout.log').read_bytes(), b'{"late":1}\n{"late":2}\n')
            self.assertEqual(raw[cutoff:], (root / 'fake-undelivered-stdout.log').read_bytes())


if __name__ == '__main__':
    unittest.main()
